%----------Nomenclature / List of Symbols--------------------------------------
\nomenclature{$^{c}\mathbf{T}_{v}$}{Homogeneous transform from vehicle frame to camera frame (\emph{vehicle$\rightarrow$camera})}
\nomenclature{$^{v}\mathbf{T}_{l}$}{Homogeneous transform from LiDAR frame to vehicle frame (\emph{LiDAR$\rightarrow$vehicle})}
\nomenclature{$\mathbf{K}_{\text{undist}}$}{Undistorted camera intrinsic matrix (pinhole model), taken from \texttt{K\_undistortion}}
\nomenclature{$s$}{OpenDRIVE longitudinal coordinate along the reference line}
\nomenclature{$\mathcal{G}$}{Road or lane connectivity graph}
\nomenclature{$\mathrm{IoU}$}{Intersection over Union used for tile matching / gating}
\nomenclature{$\mathrm{RMSE}$}{Root mean squared error}
\nomenclature{$d_H$}{Hausdorff distance}
\nomenclature{$D_{\mathrm{KL}}$}{Kullback--Leibler divergence}
\nomenclature{$\Delta t$}{Fixed simulation time step in synchronous mode}
